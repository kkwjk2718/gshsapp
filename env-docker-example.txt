# === GSHS.app Docker 배포용 환경변수 예시 ===
# 이 파일의 내용을 복사하여 서버의 .env 파일에 붙여넣으세요.

# 1. 데이터 저장 경계와 데이터베이스 (필수)
# Docker 컨테이너 내부의 명시적 writable 경로를 가리켜야 합니다.
DATA_ROOT="/app/data"
DATABASE_URL="file:/app/data/dev.db"
BACKUP_DIR="/app/data/backup"
RESTORE_ROOT="/app/data/restore"
WEATHER_CACHE_PATH="/app/data/weather-cache.json"

# 2. NextAuth 인증 (필수)
# 사용자가 접속하는 HTTPS 도메인만 사용하고 origin IP/포트를 공개 URL로 넣지 않습니다.
AUTH_URL="https://YOUR_PUBLIC_DOMAIN"
NEXTAUTH_URL="https://YOUR_PUBLIC_DOMAIN"
AUTH_TRUST_HOST="true"
# 최소 32바이트 CSPRNG 값. 예: openssl rand -base64 48
AUTH_SECRET=""
# 신뢰하는 프록시가 X-Forwarded-For를 덮어쓰는 정확한 홉 수(0-3)
TRUSTED_PROXY_HOPS="0"

# 3. NEIS API 키 (필수)
# 급식/학사일정 조회용 (기존 개발환경 키 사용 가능)
NEXT_PUBLIC_NEIS_API_KEY="your-neis-api-key"

# 4. 앱 설정 (선택/권장)
# 사이트맵 생성 등에 사용됩니다. (NEXTAUTH_URL과 동일하게 설정)
NEXT_PUBLIC_APP_URL="https://YOUR_PUBLIC_DOMAIN"

# 5. 외부 연동 (선택)
ICAL_ALLOWED_HOSTS="calendar.google.com"
BREVO_API_KEY=""
BREVO_SENDER_EMAIL=""
BREVO_SENDER_NAME="GSHS.app"
